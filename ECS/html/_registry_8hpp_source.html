422 CompID Registry::RegisterComponent(void) {
423 CompID id = GetComponentID<T>();
424 if (HasRegisterComponent<T>()) {
425 MS_WARN("Component type already registered in the registry.");
426 return id;
427 }
428
429 // else register that
430 ComponentArray<T>* carr = new(std::nothrow) ComponentArray<T>();
431 if (carr == nullptr) {
432 MS_WARN( "Failed to allocate memory for ComponentArray.");
433 return id;
434 }
435 m_typeComponentMap[id] = (IComponentArray*)carr;
436
437 return id;

447 template<typename T>
448 CompID Registry::UnregisterComponent(void) {
449 if (!HasRegisterComponent<T>()) {
450 MS_WARN("Component type not registered in the registry.");
451
452 return MultiStation::BAD_ID;
453 }
454 // else unregister that
455 CompID id = GetComponentID<T>();
456 // delete it first
457 auto it = m_typeComponentMap.find(id);
458 if (it != m_typeComponentMap.end()) {
459 delete it->second;
460 m_typeComponentMap.erase(it);
461 }
462 return id;
