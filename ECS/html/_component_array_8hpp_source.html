370 template<typename... Args>
371 T* ComponentArray<T>::AddComponent(EntID entity, Args&&... args) noexcept
372 {
373
374 // already exists -> update
375 if (m_entityToIndex.count(entity))
376 {
377 MS_WARN("Entity already has a component. Use ReplaceComponent to update it.");
378 return nullptr;
379 }
380 // new component
381 size_t newIndex = m_components.size();
382 m_components.emplace_back(std::forward<Args>(args)...);
383
384 m_entityToIndex[entity] = newIndex;
385 m_indexToEntity.push_back(entity);
386
387
388
389 return &m_components[newIndex];
390 }

410 template<typename T>
411 void ComponentArray<T>::RemoveComponent(EntID entity) noexcept {
412 if (m_components.empty()) {
413 MS_WARN("No components to remove.");
414 return;
415 }
416
417 if (!m_entityToIndex.count(entity)) { // not found entity
418 MS_WARN("Entity does not exist's.");
419 return;
420 }
421
422 size_t index = m_entityToIndex[entity];
423 size_t lastIndex = m_components.size() - 1;
424
425
426 if (index != lastIndex) {
427 m_components[index] = std::move(m_components[lastIndex]);
428 uint32_t lastEntity = m_indexToEntity[lastIndex];
429 m_entityToIndex[lastEntity] = index;
430 m_indexToEntity[index] = lastEntity;
431 }
432
433
434 // pop
435 m_components.pop_back();
436 m_indexToEntity.pop_back();
437 m_entityToIndex.erase(entity); // erase pair entity key, index vaue
438
439 }
