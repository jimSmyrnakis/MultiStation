273 void ComponentArray<T>::RemoveComponent(uint32_t entity) {
274 if (m_components.empty()) {
275 MS_ENGINE_WARN("No components to remove.");
276 return;
277 }
278
279 if (!m_entityToIndex.count(entity)) { // not found entity
280 MS_ENGINE_WARN("Entity does not exist's.");
281 return;
282 }
283
284 size_t index = m_entityToIndex[entity];
285 size_t lastIndex = m_components.size() - 1;
286
287
288 if (index != lastIndex) {
289 m_components[index] = std::move(m_components[lastIndex]);
290 uint32_t lastEntity = m_indexToEntity[lastIndex];
291 m_entityToIndex[lastEntity] = index;
292 m_indexToEntity[index] = lastEntity;
293 }
294
295
296 // pop
297 m_components.pop_back();
298 m_indexToEntity.pop_back();
299 m_entityToIndex.erase(entity); // erase pair entity key, index vaue
300
301 }
